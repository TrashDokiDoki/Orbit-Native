cmake_minimum_required(VERSION 3.21)
project(OrbitNative VERSION 0.3.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)

find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network)
add_executable(OrbitNative WIN32
    native/main.cpp native/media.cpp native/media.h native/workspace.cpp native/workspace.h native/art.qrc)
target_link_libraries(OrbitNative PRIVATE Qt6::Widgets Qt6::Network)
target_compile_definitions(OrbitNative PRIVATE ORBIT_VERSION="${PROJECT_VERSION}")
if(WIN32)
    target_link_libraries(OrbitNative PRIVATE user32)
    if(MSVC)
        target_compile_options(OrbitNative PRIVATE /utf-8)
    endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    find_package(Qt6 6.4 REQUIRED COMPONENTS DBus)
    target_link_libraries(OrbitNative PRIVATE Qt6::DBus)
endif()

enable_testing()
add_test(NAME launch COMMAND OrbitNative --smoke-test)
set_tests_properties(launch PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
option(ORBIT_WEB_SHELL "Build the experimental Orbit interface with system webviews" OFF)
if(ORBIT_WEB_SHELL)
    include(FetchContent)
    set(WEBVIEW_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    set(WEBVIEW_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(WEBVIEW_BUILD_DOCS OFF CACHE BOOL "" FORCE)
    set(WEBVIEW_INSTALL_TARGETS OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(webview
        GIT_REPOSITORY https://github.com/webview/webview.git
        GIT_TAG 0.12.0
        GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(webview)
    add_executable(OrbitShell WIN32 shell/main.cpp shell/audio.cpp shell/audio.h shell/desktop.cpp shell/desktop.h shell/network_tools.cpp shell/network_tools.h shell/shell.qrc native/media.cpp)
    target_include_directories(OrbitShell PRIVATE native)
    target_link_libraries(OrbitShell PRIVATE Qt6::Widgets Qt6::Network webview::core)
    target_compile_definitions(OrbitShell PRIVATE QT_NO_KEYWORDS ORBIT_VERSION="${PROJECT_VERSION}")
    if(WIN32)
        target_sources(OrbitShell PRIVATE native/media_windows.cpp)
        target_compile_features(OrbitShell PRIVATE cxx_std_20)
        target_compile_definitions(OrbitShell PRIVATE ORBIT_WINDOWS_MEDIA)
        target_link_libraries(OrbitShell PRIVATE windowsapp ws2_32 iphlpapi user32 comctl32 gdi32 ole32 bthprops PowrProf advapi32 shell32)
        if(MSVC)
            target_compile_options(OrbitShell PRIVATE /utf-8)
        endif()
    elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        find_package(PkgConfig REQUIRED)
        pkg_check_modules(OrbitPulse REQUIRED IMPORTED_TARGET libpulse libpulse-mainloop-glib)
        find_package(X11 REQUIRED)
        set_target_properties(OrbitShell PROPERTIES AUTOMOC ON)
        target_sources(OrbitShell PRIVATE shell/x11_input.cpp shell/portal_screenshot.cpp)
        target_link_libraries(OrbitShell PRIVATE Qt6::DBus PkgConfig::OrbitPulse X11::X11)
    endif()
    set(orbit_ui_assets
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/index.html"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/style.css"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/mascot.css"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/core.js"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/renderer.js"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/pug.png"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/icon.png"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/settings.html"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/settings.css"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/settings.js"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/capture.html"
            "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/capture.js"
            "${CMAKE_CURRENT_SOURCE_DIR}/shell/vendor/jsqr/jsQR.js"
            "${CMAKE_CURRENT_SOURCE_DIR}/shell/vendor/jsqr/LICENSE")
    add_custom_command(TARGET OrbitShell POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:OrbitShell>/orbit-shell-ui"
        COMMAND ${CMAKE_COMMAND} -E copy_if_different ${orbit_ui_assets} "$<TARGET_FILE_DIR:OrbitShell>/orbit-shell-ui")
    add_test(NAME shell-ui COMMAND OrbitShell --smoke-test)
    set_tests_properties(shell-ui PROPERTIES TIMEOUT 40)
endif()

option(ORBIT_PACKAGE_SHELL "Package the WebView preview separately from the Qt Widgets edition" OFF)
if(ORBIT_PACKAGE_SHELL AND NOT ORBIT_WEB_SHELL)
    message(FATAL_ERROR "ORBIT_PACKAGE_SHELL requires ORBIT_WEB_SHELL")
endif()
set(orbit_install_target OrbitNative)
set(orbit_install_name "Orbit Native")
set(orbit_package_id orbit-native)
set(orbit_package_prefix OrbitNative)
set(orbit_desktop_file packaging/orbit-native.desktop)
if(ORBIT_PACKAGE_SHELL)
    set(orbit_install_target OrbitShell)
    set(orbit_install_name "Orbit Native WebView Preview")
    set(orbit_package_id orbit-native-shell)
    set(orbit_package_prefix OrbitShell)
    set(orbit_desktop_file packaging/orbit-native-shell.desktop)
    install(FILES ${orbit_ui_assets} DESTINATION bin/orbit-shell-ui)
endif()

install(TARGETS ${orbit_install_target} RUNTIME DESTINATION bin)

set(CPACK_PACKAGE_NAME "${orbit_package_prefix}")
set(CPACK_PACKAGE_VENDOR "Orbit")
set(CPACK_PACKAGE_CONTACT "Orbit Native contributors <orbit@users.noreply.github.com>")
set(CPACK_PACKAGE_HOMEPAGE_URL "https://github.com/TrashDokiDoki/Orbit-Native")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Native desktop island and launcher")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/resources/app/LICENSE")
if(WIN32)
    qt_generate_deploy_app_script(TARGET ${orbit_install_target} OUTPUT_SCRIPT deploy_script)
    install(SCRIPT "${deploy_script}")
    if(MSVC)
        set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION bin)
        include(InstallRequiredSystemLibraries)
    endif()
    install(FILES resources/app/LICENSE DESTINATION ".")
    set(CPACK_GENERATOR NSIS)
    set(orbit_windows_arch x64)
    if(CMAKE_GENERATOR_PLATFORM MATCHES "^[Aa][Rr][Mm]64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "^(ARM64|aarch64)$")
        set(orbit_windows_arch arm64)
    endif()
    set(CPACK_PACKAGE_FILE_NAME "${orbit_package_prefix}-${PROJECT_VERSION}-windows-${orbit_windows_arch}")
    set(CPACK_NSIS_DISPLAY_NAME "${orbit_install_name}")
    set(CPACK_PACKAGE_INSTALL_DIRECTORY "${orbit_install_name}")
    set(CPACK_PACKAGE_EXECUTABLES "${orbit_install_target}" "${orbit_install_name}")
    set(CPACK_NSIS_MUI_FINISHPAGE_RUN "${orbit_install_target}.exe")
    set(CPACK_NSIS_ENABLE_UNINSTALL_BEFORE_INSTALL ON)
    if(ORBIT_PACKAGE_SHELL)
        set(ORBIT_TESSERACT_BUNDLE "" CACHE PATH "Path to the packaged Windows OCR runtime")
        if(ORBIT_TESSERACT_BUNDLE)
            foreach(ocr_file tesseract.exe tessdata/eng.traineddata tessdata/rus.traineddata)
                if(NOT EXISTS "${ORBIT_TESSERACT_BUNDLE}/${ocr_file}")
                    message(FATAL_ERROR "Incomplete Windows OCR bundle: ${ocr_file}")
                endif()
            endforeach()
            install(DIRECTORY "${ORBIT_TESSERACT_BUNDLE}/" DESTINATION bin/orbit-ocr)
        endif()
        set(ORBIT_WEBVIEW2_BOOTSTRAPPER "" CACHE FILEPATH "Microsoft Evergreen WebView2 bootstrapper")
        if(NOT EXISTS "${ORBIT_WEBVIEW2_BOOTSTRAPPER}")
            message(FATAL_ERROR "Set ORBIT_WEBVIEW2_BOOTSTRAPPER to the signed Microsoft Evergreen bootstrapper")
        endif()
        file(TO_NATIVE_PATH "${ORBIT_WEBVIEW2_BOOTSTRAPPER}" orbit_webview_bootstrapper_native)
        file(TO_NATIVE_PATH "${CMAKE_CURRENT_BINARY_DIR}/orbit-webview2.nsh" orbit_webview_include_native)
        configure_file(packaging/webview2.nsh.in "${CMAKE_CURRENT_BINARY_DIR}/orbit-webview2.nsh" @ONLY)
        set(CPACK_NSIS_EXTRA_PREINSTALL_COMMANDS "!include \"${orbit_webview_include_native}\"")
    endif()
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    install(FILES "${orbit_desktop_file}" DESTINATION share/applications)
    install(FILES resources/app/icon.png DESTINATION share/icons/hicolor/256x256/apps RENAME ${orbit_package_id}.png)
    install(FILES resources/app/LICENSE DESTINATION share/doc/${orbit_package_id} RENAME copyright)
    set(CPACK_GENERATOR DEB)
    set(CPACK_DEBIAN_PACKAGE_NAME ${orbit_package_id})
    set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
    set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
    set(CPACK_DEBIAN_PACKAGE_DEPENDS "qt6-qpa-plugins")
    if(ORBIT_PACKAGE_SHELL)
        string(APPEND CPACK_DEBIAN_PACKAGE_DEPENDS ", tesseract-ocr, tesseract-ocr-rus")
    endif()
endif()
set(CPACK_VERBATIM_VARIABLES TRUE)
include(CPack)
